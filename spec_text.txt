4. End-to-End Data Flow & Sequence
Sequence (Summary):
1. Client requests SAS or uploads via FastAPI.
2. Blob Storage receives file → Event Grid publishes → Service Bus enqueues message.
3. Worker downloads blob, preprocesses, calls Form Recognizer for extraction.
4. Worker validates; auto-accept or create review task.
5. On accept: persist to DB, index to Search, move blob to processed, integrate downstream.
6. On review: reviewer edits via Next.js UI; FastAPI updates DB & search; then integration.
7. Audit trail appended.

// frontend/pages/review/[id].js
import { useState, useEffect } from "react";
import { useRouter } from "next/router";
export default function ReviewPage() {
  const router = useRouter();
  const { id } = router.query;
  const [doc, setDoc] = useState(null);
  useEffect(() => {
    if (!id) return;
    fetch(`/api/documents/${id}`).then(r => r.json()).then(setDoc);
  }, [id]);
  if (!doc) return <div>Loading...</div>;
  return (
    <div className="p-4">
      <h1 className="text-xl font-bold">Review {doc.filename}</h1>
      <div className="mt-4">
        <img src={doc.previewUrl} alt="preview" className="border" />
      </div>
      <form>
        <label className="block mt-2">Vendor</label>
        <input defaultValue={doc.extractedFields.vendorName} className="border p-2" />
      </form>
    </div>
  );
}
